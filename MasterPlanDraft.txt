3.2 Custom 3D Golden Element

```tsx
// src/components/3d/GoldenGlobe.tsx
import { useRef } from 'react';
import { useFrame } from '@react-three/fiber';
import { Mesh } from 'three';

export const GoldenGlobe = () => {
  const meshRef = useRef<Mesh>(null!);
  
  useFrame((state) => {
    meshRef.current.rotation.y += 0.003;
    meshRef.current.rotation.x = Math.sin(state.clock.elapsedTime * 0.2) * 0.1;
  });
  
  return (
    <mesh ref={meshRef}>
      <sphereGeometry args={[1.2, 64, 64]} />
      <meshStandardMaterial 
        color="#C9A96E" 
        metalness={0.8} 
        roughness={0.2}
        emissive="#8B7D3C"
        emissiveIntensity={0.1}
      />
    </mesh>
  );
};
```

🔐 Phase 4: Authentication & Dashboard (Week 3)

4.1 Auth Hook

```typescript
// src/hooks/useAuth.ts
import { useState, useEffect } from 'react';
import { auth } from '../firebase/config';
import { onAuthStateChanged, User } from 'firebase/auth';

export const useAuth = () => {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (user) => {
      setUser(user);
      setLoading(false);
    });
    return unsubscribe;
  }, []);

  return { user, loading };
};
```

4.2 Protected Dashboard Route

```tsx
// src/pages/Dashboard.tsx
import { useAuth } from '../hooks/useAuth';
import { ResidentList } from '../components/dashboard/ResidentList';
import { CareLogForm } from '../components/dashboard/CareLogForm';

export const Dashboard = () => {
  const { user, loading } = useAuth();
  
  if (loading) return <div>Loading...</div>;
  if (!user) return <div>Please sign in</div>;
  
  return (
    <div className="grid grid-cols-1 gap-6 p-6 md:grid-cols-2">
      <ResidentList />
      <CareLogForm />
    </div>
  );
};
```

⚡ Efficiency Tips & Pointers

1. Use Vite Instead of CRA

Vite offers instant server start and lightning-fast HMR. Your dev server will boot in < 500ms vs 5-10s with Create React App.

2. Lazy Load 3D Components

```tsx
const Scene = lazy(() => import('./components/3d/Scene'));
```

This prevents the 3D bundle from blocking initial page load.

3. Optimize 3D Performance

· Use @react-three/drei helpers like Html for overlays
· Reduce polygon count on models
· Use useMemo for geometry and materials
· Enable shadows only when needed

4. Firestore Security Rules (Critical!)

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /residents/{resident} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
        request.auth.token.email.matches('.*@goldenmomentscare.com$');
    }
    match /careLogs/{log} {
      allow read, write: if request.auth != null;
    }
  }
}
```

5. Environment Variables

Create .env.local:

```
VITE_FIREBASE_API_KEY=xxx
VITE_FIREBASE_AUTH_DOMAIN=xxx
VITE_FIREBASE_PROJECT_ID=xxx
```

6. Use TanStack Router for Type-Safe Routing

```bash
npm install @tanstack/react-router
```

7. Service Worker Caching Strategy

The vite-plugin-pwa with Workbox provides advanced caching. Configure it to cache:

· Static assets (immediate)
· API responses (stale-while-revalidate)
· 3D models (cache-first)
